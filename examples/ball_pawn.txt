# Classic Cornell box, but with both of the usual boxes replaced by a sphere textured
# with the famous Pixar ball (assets/pixar_ball.png, loaded as an LDR pigment
# via image(STRING, factor_a, avr_lum, gamma)) and a
# golden chess pawn mesh. The red/green walls still bleed
# color onto both, so global illumination stays the point of the scene.
#
# Suggested render command (higher quality than the CLI defaults, since the
# path tracer has no direct-light sampling and relies purely on GI bounces).
# --reflectance-policy rescale is required: the ball photo has near-saturated
# pixels whose LDR-to-HDR inverse tone-mapping lands above 1.0, which the
# default `reject` policy would otherwise refuse to render. Add --release, the
# pawn mesh makes it matter:
#   cargo run --release -- render examples/ball_pawn.txt --num-of-rays 8 --max-depth 5 --antialiasing 8 \
# --reflectance-policy rescale --pfm-output ball_pawn --image-output ball_pawn

material white_material(
    uniform(<0.73, 0.73, 0.73>),
    diffuse(),
    uniform(black)
)

material red_material(
    uniform(<0.65, 0.05, 0.05>),
    diffuse(),
    uniform(black)
)

material green_material(
    uniform(<0.12, 0.45, 0.15>),
    diffuse(),
    uniform(black)
)

material light_material(
    uniform(white),
    diffuse(),
    uniform(<15, 15, 15>)
)

material ball_material(
    image("assets/pixar_ball.png", 0.18, 1.0, 2.2),
    diffuse(),
    uniform(black)
)

material pawn_material(
    uniform(<0.83, 0.68, 0.21>),
    specular(),
    uniform(black)
)

# Floor (z = 0)
plane(white_material, identity, true)

# Ceiling (z = 2)
plane(white_material, translation([0, 0, 2]))

# Back wall (x = 4)
plane(white_material, translation([4, 0, 0]) * rotation_y(90))

# Left wall, red (y = 1)
plane(red_material, translation([0, 1, 0]) * rotation_x(90))

# Right wall, green (y = -1)
plane(green_material, translation([0, -1, 0]) * rotation_x(90))

# Textured ball, in place of the short box (near the green wall, resting on the floor)
sphere(ball_material, translation([1.45, -0.2, 0.35]) * scaling(0.35) * rotation_y(30) * rotation_z(60))

# Golden pawn, in place of the tall box (near the red wall). scaling is applied
# first so its base stays anchored on z = 0, then translation places it.
simple_mesh(pawn_material, "assets/pawn.obj", translation([2.6, 0.4, 0.0]) * scaling(0.4))

# Ceiling light panel, just below the ceiling
box(light_material, point([1.5, -0.4, 1.97]), point([2.5, 0.4, 1.999]))

camera(
    perspective,
    translation([0, 0, 1]),
    2.5
)
