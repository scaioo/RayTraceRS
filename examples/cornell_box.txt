# Classic Cornell box: a small room lit by a single emissive patch on the
# ceiling, used to showcase global illumination (color bleeding between the
# red/green walls and the two white boxes) with the path tracer.
#
# Suggested render command (higher quality than the CLI defaults, since the
# path tracer has no direct-light sampling and relies purely on GI bounces):
#   cargo run -- render examples/cornell_box.txt --num-of-rays 25 --max-depth 6 --antialiasing 8

material white_material(
    uniform(<0.73, 0.73, 0.73>),
    diffuse(),
    uniform(black)
)

material red_material(
    uniform(<0.65, 0.05, 0.05>),
    diffuse(),
    uniform(black)
)

material green_material(
    uniform(<0.12, 0.45, 0.15>),
    diffuse(),
    uniform(black)
)

material light_material(
    uniform(white),
    diffuse(),
    uniform(<15, 15, 15>)
)

# Floor (z = 0)
plane(white_material, identity, true)

# Ceiling (z = 2)
plane(white_material, translation([0, 0, 2]))

# Back wall (x = 4)
plane(white_material, translation([4, 0, 0]) * rotation_y(90))

# Left wall, red (y = 1)
plane(red_material, translation([0, 1, 0]) * rotation_x(90))

# Right wall, green (y = -1)
plane(green_material, translation([0, -1, 0]) * rotation_x(90))

# Short box
aabb(white_material, point([1.0, -0.55, 0.0]), point([1.9, 0.15, 0.65]))

# Tall box
aabb(white_material, point([2.3, 0.15, 0.0]), point([3.1, 0.85, 1.5]))

# Ceiling light panel, just below the ceiling
aabb(light_material, point([1.5, -0.4, 1.97]), point([2.5, 0.4, 1.999]))

camera(
    perspective,
    translation([-0.8, 0, 1]),
    2.5
)
