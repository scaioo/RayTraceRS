# =====================================================================
#  Greek temple at sunset  --  generated for the `cylinder` branch.
#  Temple:  elongated 6 x 12 peristyle (hexastyle facade, 12 deep).
#  Columns: stacked tapering cylinders (drums) + echinus + abacus box.
#  Base:    stepped crepidoma made of axis-aligned boxes.
#  Roof:    triangle mesh (assets/greek_temple_roof.obj).
#  Sky:     emissive gradient dome, vivid red/orange sunset, which also
#           lights the scene.
#
#  Use the path tracer so the emissive sky acts as soft area light.
#
#  Quick preview (~10 s):
#    cargo run --release -- render examples/greek_temple.txt \
#      --width 480 --height 320 --num-of-rays 3 --antialiasing 3
#
#  Quality render (~1.5 min; ~280 objects x a branching path tracer,
#  so keep max-depth low -- depth 4 gets very slow):
#    cargo run --release -- render examples/greek_temple.txt \
#      --width 900 --height 600 --num-of-rays 3 --max-depth 3 \
#      --antialiasing 6
# =====================================================================

# ------------------------------- materials -------------------------------
material marble(
    uniform(<0.86, 0.82, 0.73>),
    diffuse(),
    uniform(black)
)

material stone(
    uniform(<0.58, 0.56, 0.50>),
    diffuse(),
    uniform(black)
)

material roof_material(
    uniform(<0.60, 0.40, 0.30>),
    diffuse(),
    uniform(black)
)

material ground_material(
    uniform(<0.27, 0.25, 0.21>),
    diffuse(),
    uniform(black)
)

# Sky dome: no reflectance, radiance comes from a vertical gradient.
# NOTE: the renderer feeds get_color the *oriented* uv (u,v)->(v,-u), so
# a gradient that varies with elevation must use angle 0 (which then reads
# the sphere's polar coordinate v).  color1 = zenith (deep magenta-violet),
# color2 = nadir (fiery orange-red); the visible sky (v~0.26..0.5) burns
# from a dark red top down to an intense orange glow at the horizon.
material sky_material(
    uniform(black),
    diffuse(),
    gradient(<0.16, 0.03, 0.28>, <3.80, 0.90, 0.12>, 0)
)

# Setting sun: a bright warm emitter acting as a soft area light.
material sun_material(
    uniform(black),
    diffuse(),
    uniform(<13.0, 9.0, 4.5>)
)

# ------------------------------- sky & ground ----------------------------
sphere(sky_material, scaling([100, 100, 100]))
sphere(sun_material, translation([-3.0, -22.0, 3.5]) * scaling(5.0))
plane(ground_material, identity)

# --- Realistic textured floor (optional) ---------------------------------

# lines above for something like:
#
#   material ground_material(
#       image("assets/ground.jpg", 0.18, 1.0, 2.2),   # png/jpg: a, lum, gamma
#       diffuse(),
#       uniform(black)
#   )
#   plane(ground_material, scaling([3, 3, 1]))         # 1 tile every 3 units
#

#   checkered(<0.42, 0.40, 0.35>, <0.34, 0.32, 0.28>, 40)   # + plane(..., true)
