# Image-textured sphere: an HDR (.pfm) image mapped onto a sphere as a diffuse
# pigment, on a ground plane under an emissive sky.
#
# The texture's values exceed the [0,1] reflectance range, so rescale them:
#   cargo run -- render examples/issue_world.txt --reflectance-policy rescale --antialiasing 4

material sky_material(
    uniform(<0.5, 0.9, 1.0>),
    diffuse(),
    uniform(<0.5, 0.9, 1.0>)
)

material ground_material(
    uniform(<0.3, 0.4, 0.8>),
    diffuse(),
    uniform(black)
)

material sphere_material(
    image("tests/assets/memorial.pfm"),
    diffuse(),
    uniform(black)
)

sphere(sky_material, scaling([200.0, 200.0, 200.0]) * translation([0.0, 0.0, 0.4]))
plane(ground_material, identity, true)
sphere(sphere_material, translation([0.0, 0.0, 1.0]))

camera(
perspective,
translation([-1, 0, 1]),
1.0
)
