# Example scene demonstrating LDR (PNG/JPEG) textures as pigments.
# The "ball_material" pigment loads a PNG file and reconstructs an
# approximate HDR texture from it via image(STRING, factor_a, avr_lum, gamma).

material sky_material(
    uniform(<0.5, 0.9, 1.0>),
    diffuse(),
    uniform(<0.5, 0.9, 1.0>)
)

material ground_material(
    uniform(<0.3, 0.4, 0.8>),
    diffuse(),
    uniform(black)
)

material ball_material(
    image("assets/pixar_ball.png", 0.18, 1.0, 2.2),
    diffuse(),
    uniform(black)
)

sphere(sky_material, scaling([200.0, 200.0, 200.0]) * translation([0.0, 0.0, 0.4]))
plane(ground_material, identity, true)
sphere(ball_material, translation([0.0, 0.0, 1.0]))

camera(
perspective,
translation([-3, 0, 1]),
1.0
)
