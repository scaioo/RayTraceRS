# ============================================================
# Scena di test: CSG con operazione union + due CSG con difference,
# pavimento checkered, point light.
#
# Keyword verificate contro lexer.rs (parse_identifier_token):
# material, plane, sphere, box, csg, simple_mesh, diffuse, specular,
# uniform, checkered, image, gradient, identity, translation,
# rotation_x/y/z, scaling, camera, orthogonal, perspective, float,
# point, point_light, spherical_light, union/u, difference/diff/d,
# intersection/intr, true/false, black/white.
# ============================================================

# --- Materiali ---

material floor_material(
    checkered(<0.85, 0.85, 0.85>, <0.15, 0.15, 0.15>, 8),
    diffuse(),
    uniform(<0, 0, 0>)
)

material red_material(
    uniform(<0.8, 0.1, 0.1>),
    diffuse(),
    uniform(<0, 0, 0>)
)

material blue_material(
    uniform(<0.1, 0.1, 0.8>),
    diffuse(),
    uniform(<0, 0, 0>)
)

material green_material(
    uniform(<0.1, 0.7, 0.2>),
    diffuse(),
    uniform(<0, 0, 0>)
)

material yellow_material(
    uniform(<0.85, 0.75, 0.1>),
    diffuse(),
    uniform(<0, 0, 0>)
)

material purple_material(
    uniform(<0.55, 0.15, 0.65>),
    diffuse(),
    uniform(<0, 0, 0>)
)

material cyan_material(
    uniform(<0.1, 0.7, 0.7>),
    diffuse(),
    uniform(<0, 0, 0>)
)

# --- Pavimento checkered ---
# procedural_texture=true: senza questo flag il checkered pattern
# viene mappato su UV [0,1] e si vede una sola cella (bug osservato:
# pavimento tutto bianco).

plane(floor_material, translation([0, 0, -1]), true)

# --- CSG 1: UNION di due sfere che si intersecano ---

csg(
    union,
    sphere(red_material, translation([-2.0, 2.0, 3.3]) * scaling(1.1)),
    sphere(blue_material, translation([-1.1, 2.0, 3.3]) * scaling(1.1))
)

# --- CSG 2: DIFFERENCE di due sfere che si intersecano ---

csg(
    difference,
    sphere(green_material, translation([0.8, 2.0, 0.3]) * scaling(1.2)),
    sphere(yellow_material, translation([1.5, 2.0, 0.5]) * scaling(0.9))
)

# --- CSG 3: altre due sfere in DIFFERENCE ---

csg(
    union,
    sphere(purple_material, translation([4.2, 4.0, 3.3]) * scaling(1.2)),
    sphere(cyan_material, translation([4.8, 4.0, 3.6]) * scaling(0.9))
)

# --- Point light ---

point_light(point([0, -3, 5]), white)

# --- Camera ---
# rotation_z(90) prima della translation: ruota l'orientamento locale
# della camera (di default guarda verso +x) in modo che punti verso
# +y, dove si trovano le sfere/CSG. L'ordine conta: si applica prima
# la rotazione "sul posto" e poi si sposta la camera già ruotata.

camera(perspective, translation([0, -6, 1.5]) * rotation_z(90), 1.5)
